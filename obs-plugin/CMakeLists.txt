cmake_minimum_required(VERSION 3.25)
project(vc-mix VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This build supports 64-bit Windows OBS.")
endif()
if(NOT OBS_SOURCE_DIR OR NOT OBS_IMPORT_DIR)
  message(FATAL_ERROR "Run scripts/build-obs-plugin.ps1 to prepare the OBS SDK.")
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
add_library(obs-sdk INTERFACE)
target_include_directories(obs-sdk SYSTEM INTERFACE "${OBS_SOURCE_DIR}/libobs" "${OBS_SOURCE_DIR}/frontend/api" "${OBS_IMPORT_DIR}")
target_link_libraries(obs-sdk INTERFACE "${OBS_IMPORT_DIR}/obs.lib" "${OBS_IMPORT_DIR}/obs-frontend-api.lib")
target_compile_definitions(obs-sdk INTERFACE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(vc-mix MODULE
  src/plugin-main.cpp src/ReceiverBridge.cpp src/ParticipantSource.cpp src/SettingsStore.cpp src/SettingsDialog.cpp)
target_link_libraries(vc-mix PRIVATE obs-sdk crypt32 Qt6::Widgets)
target_compile_options(vc-mix PRIVATE /W4 /utf-8 /permissive-)
target_link_options(vc-mix PRIVATE "/PDBALTPATH:vc-mix.pdb")
set_target_properties(vc-mix PROPERTIES PREFIX "")
add_executable(vc-mix-integration
  tests/integration.cpp src/ReceiverBridge.cpp src/ParticipantSource.cpp src/SettingsStore.cpp)
target_link_libraries(vc-mix-integration PRIVATE obs-sdk crypt32)
target_compile_options(vc-mix-integration PRIVATE /W4 /utf-8 /permissive-)
add_executable(vc-mix-settings-test
  tests/settings-window.cpp src/SettingsDialog.cpp src/ReceiverBridge.cpp src/SettingsStore.cpp)
target_link_libraries(vc-mix-settings-test PRIVATE obs-sdk crypt32 Qt6::Widgets)
target_compile_options(vc-mix-settings-test PRIVATE /W4 /utf-8 /permissive-)
